cmake_minimum_required(VERSION 3.24)
project(cuda_ch36_distributed LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
add_executable(partition_selftest partition_selftest.cpp)
enable_testing()
add_test(NAME ch36_partition_selftest COMMAND partition_selftest)

find_package(MPI COMPONENTS CXX QUIET)
if(MPI_CXX_FOUND)
  add_executable(mpi_halo_heat mpi_halo_heat.cu)
  target_link_libraries(mpi_halo_heat PRIVATE MPI::MPI_CXX)
  target_compile_options(mpi_halo_heat PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
  find_path(NCCL_INCLUDE_DIR nccl.h)
  find_library(NCCL_LIBRARY NAMES nccl)
  if(NCCL_INCLUDE_DIR AND NCCL_LIBRARY)
    add_executable(nccl_collectives nccl_collectives.cu)
    target_include_directories(nccl_collectives PRIVATE ${NCCL_INCLUDE_DIR})
    target_link_libraries(nccl_collectives PRIVATE MPI::MPI_CXX ${NCCL_LIBRARY})
    target_compile_options(nccl_collectives PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
  else()
    message(STATUS "NCCL headers/library absent: nccl_collectives target omitted")
  endif()
else()
  message(STATUS "MPI C++ absent: MPI and NCCL example targets omitted")
endif()

cmake_minimum_required(VERSION 3.24)
project(cuda_ch16_matmul_case LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
add_executable(matmul_compare matmul_compare.cu)
target_link_libraries(matmul_compare PRIVATE CUDA::cudart_static)
if(TARGET CUDA::cublas)
    target_link_libraries(matmul_compare PRIVATE CUDA::cublas)
    target_compile_definitions(matmul_compare PRIVATE GUIDE_HAS_CUBLAS=1)
else()
    target_compile_definitions(matmul_compare PRIVATE GUIDE_HAS_CUBLAS=0)
    message(STATUS "cuBLAS development library unavailable; SGEMM comparison will report SKIP")
endif()
target_compile_options(matmul_compare PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
enable_testing()
add_test(NAME ch16_matmul_compare COMMAND matmul_compare)

cmake_minimum_required(VERSION 3.24)
project(cuda_ch35_multi_gpu LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
add_executable(multi_gpu_images multi_gpu_images.cu)
target_compile_options(multi_gpu_images PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
enable_testing()
add_test(NAME ch35_partition_selftest COMMAND multi_gpu_images --selftest)
add_test(NAME ch35_two_gpu_images COMMAND multi_gpu_images)
set_tests_properties(ch35_two_gpu_images PROPERTIES SKIP_RETURN_CODE 77)

cmake_minimum_required(VERSION 3.24)
project(cuda_ch37_architecture_optimization LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
add_executable(async_stencil async_stencil.cu)
target_compile_options(async_stencil PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo> $<$<COMPILE_LANGUAGE:CUDA>:-Xptxas=-v>)
enable_testing()
add_test(NAME ch37_async_stencil COMMAND async_stencil)

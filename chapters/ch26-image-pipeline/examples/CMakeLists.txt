cmake_minimum_required(VERSION 3.24)
project(cuda_ch26_image_pipeline LANGUAGES CXX CUDA)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
add_executable(image_pipeline image_pipeline.cu)
target_include_directories(image_pipeline PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../common)
target_compile_options(image_pipeline PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
enable_testing()
add_test(NAME ch26_image_pipeline COMMAND image_pipeline ${CMAKE_CURRENT_BINARY_DIR}/ch26-output)

cmake_minimum_required(VERSION 3.24)
project(cuda_ch33_stencil_heat LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
add_executable(stencil_heat stencil_heat.cu)
target_compile_options(stencil_heat PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
enable_testing()
add_test(NAME ch33_stencil_heat COMMAND stencil_heat)

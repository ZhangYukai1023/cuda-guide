cmake_minimum_required(VERSION 3.24)
project(cuda_ch28_softmax_normalization LANGUAGES CXX CUDA)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
add_executable(row_normalization row_normalization.cu)
add_executable(mixed_precision_softmax mixed_precision_softmax.cu)
target_compile_options(row_normalization PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
target_compile_options(mixed_precision_softmax PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
enable_testing()
add_test(NAME ch28_row_normalization COMMAND row_normalization)
add_test(NAME ch28_mixed_precision_softmax COMMAND mixed_precision_softmax)

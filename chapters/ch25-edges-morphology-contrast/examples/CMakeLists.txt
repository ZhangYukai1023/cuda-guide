cmake_minimum_required(VERSION 3.24)
project(cuda_ch25_edges_morphology_contrast LANGUAGES CXX CUDA)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
add_executable(image_features image_features.cu)
target_include_directories(image_features PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../common)
target_compile_options(image_features PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
enable_testing()
add_test(NAME ch25_image_features COMMAND image_features ${CMAKE_CURRENT_BINARY_DIR}/ch25-images)

cmake_minimum_required(VERSION 3.24)
project(cuda_ch38_integrated_projects LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
add_executable(project_b_operators operators/main.cu operators/operators.cu)
add_executable(project_c_heat heat/heat_delivery.cu)
target_compile_options(project_b_operators PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
target_compile_options(project_c_heat PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
enable_testing()
add_test(NAME ch38_project_b COMMAND project_b_operators)
add_test(NAME ch38_project_c COMMAND project_c_heat)

cmake_minimum_required(VERSION 3.24)
project(cuda_ch31_linear_sparse_solvers LANGUAGES CXX CUDA)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
find_package(CUDAToolkit REQUIRED)
enable_testing()
add_executable(reference_solvers reference_solvers.cu)
target_link_libraries(reference_solvers PRIVATE CUDA::cudart_static)
target_compile_options(reference_solvers PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
add_test(NAME ch31_reference_solvers COMMAND reference_solvers)
if(TARGET CUDA::cublas AND TARGET CUDA::cusparse AND TARGET CUDA::cusolver)
  add_executable(linear_solvers linear_solvers.cu)
  target_link_libraries(linear_solvers PRIVATE CUDA::cublas CUDA::cusparse CUDA::cusolver)
  target_compile_options(linear_solvers PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
  add_test(NAME ch31_linear_solvers COMMAND linear_solvers)
else()
  message(STATUS "cuBLAS/cuSPARSE/cuSOLVER development libraries unavailable: library solver target not built")
endif()

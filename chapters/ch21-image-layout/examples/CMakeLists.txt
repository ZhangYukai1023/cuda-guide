cmake_minimum_required(VERSION 3.24)
project(cuda_ch21_image_layout LANGUAGES CXX CUDA)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
add_executable(image_layout image_layout.cu)
target_include_directories(image_layout PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../common)
target_compile_options(image_layout PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
enable_testing()
add_test(NAME ch21_image_layout COMMAND image_layout ${CMAKE_CURRENT_BINARY_DIR}/ch21-images)

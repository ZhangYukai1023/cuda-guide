cmake_minimum_required(VERSION 3.24)
project(cuda_ch10_debugging LANGUAGES CXX CUDA)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
add_executable(debug_cases debug_cases.cu)
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    target_compile_options(debug_cases PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-G> $<$<COMPILE_LANGUAGE:CUDA>:-g>)
else()
    target_compile_options(debug_cases PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endif()
enable_testing()
add_test(NAME ch10_debug_safe COMMAND debug_cases safe)

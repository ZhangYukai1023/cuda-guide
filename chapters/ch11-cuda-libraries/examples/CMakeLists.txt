cmake_minimum_required(VERSION 3.24)
project(cuda_ch11_libraries LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
add_executable(library_baselines library_baselines.cu)
target_link_libraries(library_baselines PRIVATE CUDA::cudart_static)
if(TARGET CUDA::cublas)
    target_link_libraries(library_baselines PRIVATE CUDA::cublas)
    target_compile_definitions(library_baselines PRIVATE GUIDE_HAS_CUBLAS=1)
else()
    target_compile_definitions(library_baselines PRIVATE GUIDE_HAS_CUBLAS=0)
    message(STATUS "cuBLAS development library unavailable; its example will report SKIP")
endif()
target_compile_options(library_baselines PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
enable_testing()
add_test(NAME ch11_library_baselines COMMAND library_baselines)

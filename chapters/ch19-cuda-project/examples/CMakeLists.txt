cmake_minimum_required(VERSION 3.24)
project(cuda_ch19_project LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
add_library(cuda_array_ops STATIC
    src/array_ops.cu src/array_kernel.cu src/device_math.cu)
target_include_directories(cuda_array_ops PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(cuda_array_ops PUBLIC CUDA::cudart_static)
set_target_properties(cuda_array_ops PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
add_executable(array_module_demo src/main.cu)
target_link_libraries(array_module_demo PRIVATE cuda_array_ops)
set_target_properties(array_module_demo PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
enable_testing()
add_test(NAME ch19_array_module COMMAND array_module_demo)

cmake_minimum_required(VERSION 3.24)
project(cuda_ch10 LANGUAGES CXX CUDA)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
add_executable(multi_gpu multi_gpu.cu)
target_compile_options(multi_gpu PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
enable_testing()
add_test(NAME ch10_multi_gpu COMMAND multi_gpu)
add_test(NAME ch10_two_devices COMMAND multi_gpu --require-two)
set_tests_properties(ch10_two_devices PROPERTIES SKIP_RETURN_CODE 77)

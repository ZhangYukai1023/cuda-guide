cmake_minimum_required(VERSION 3.24)
project(cuda_ch32_fft_frequency LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
find_package(CUDAToolkit REQUIRED)
enable_testing()
add_executable(direct_frequency direct_frequency.cu)
target_link_libraries(direct_frequency PRIVATE CUDA::cudart_static)
target_compile_options(direct_frequency PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
add_test(NAME ch32_direct_frequency COMMAND direct_frequency)
if(TARGET CUDA::cufft)
  add_executable(fft_frequency fft_frequency.cu)
  target_link_libraries(fft_frequency PRIVATE CUDA::cufft)
  target_compile_options(fft_frequency PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
  add_test(NAME ch32_fft_frequency COMMAND fft_frequency)
else()
  message(STATUS "cuFFT development library unavailable: cuFFT example not built")
endif()

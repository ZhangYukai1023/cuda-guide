cmake_minimum_required(VERSION 3.24)
project(cuda_ch29_tensor_core_cutlass LANGUAGES CXX CUDA)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
find_package(CUDAToolkit REQUIRED)
add_executable(tensor_core_gemm tensor_core_gemm.cu)
target_link_libraries(tensor_core_gemm PRIVATE CUDA::cudart_static)
if(TARGET CUDA::cublas)
  target_link_libraries(tensor_core_gemm PRIVATE CUDA::cublas)
  target_compile_definitions(tensor_core_gemm PRIVATE GUIDE_HAS_CUBLAS=1)
else()
  target_compile_definitions(tensor_core_gemm PRIVATE GUIDE_HAS_CUBLAS=0)
  message(STATUS "cuBLAS development library unavailable: GEMMEx comparison will report SKIP")
endif()
target_compile_options(tensor_core_gemm PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
enable_testing()
add_test(NAME ch29_tensor_core_gemm COMMAND tensor_core_gemm)
set_tests_properties(ch29_tensor_core_gemm PROPERTIES SKIP_RETURN_CODE 77)
if(DEFINED CUTLASS_ROOT AND EXISTS "${CUTLASS_ROOT}/include/cutlass/gemm/device/gemm.h")
  add_executable(cutlass_gemm cutlass_gemm.cu)
  target_include_directories(cutlass_gemm PRIVATE "${CUTLASS_ROOT}/include")
  target_link_libraries(cutlass_gemm PRIVATE CUDA::cudart_static)
  target_compile_options(cutlass_gemm PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
  add_test(NAME ch29_cutlass_gemm COMMAND cutlass_gemm)
else()
  message(STATUS "CUTLASS_ROOT absent: optional CUTLASS comparison is not built")
endif()

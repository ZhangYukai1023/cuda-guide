cmake_minimum_required(VERSION 3.24)
project(cuda_ch13_nsight LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
add_executable(profile_cases profile_cases.cu)
target_include_directories(profile_cases PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
target_link_libraries(profile_cases PRIVATE CUDA::cudart_static ${CMAKE_DL_LIBS})
target_compile_options(profile_cases PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
enable_testing()
add_test(NAME ch13_profile_tiny COMMAND profile_cases tiny)
add_test(NAME ch13_profile_batch COMMAND profile_cases batch)
add_test(NAME ch13_profile_stride COMMAND profile_cases stride)
